# 知间 AI（KnowFlow）— 模块一：基础模块开发文档

> 对应规格书「开发阶段」第 1、2 阶段，实现项目初始化与资料管理闭环。
> 目标：跑通「初始化 → 数据模型 → 资料 CRUD → 首页」的最基础可用闭环，刷新后数据不丢失。

## 6. 工具函数（src/utils/）

### 6.1 uuid.ts

```ts
export function uuid(): string {
  return crypto.randomUUID
    ? crypto.randomUUID()
    : "xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx".replace(/[xy]/g, (c) => {
        const r = (Math.random() * 16) | 0;
        const v = c === "x" ? r : (r & 0x3) | 0x8;
        return v.toString(16);
      });
}
```

### 6.2 validation.ts

```ts
export function validateDocument(input: { title: string; content: string }) {
  const errors: string[] = [];
  if (!input.title || !input.title.trim()) {
    errors.push("标题不能为空");
  }
  if (!input.content || !input.content.trim()) {
    errors.push("正文为空，建议补充内容后再保存");
  }
  return { valid: errors.length === 0, errors };
}
```

## 7. 业务服务层（src/services/）

> 原则：所有 CRUD 逻辑放进 Service，Pinia store 只做状态分发，禁止在页面组件里直接操作 Dexie。

### 7.1 documentService.ts

```ts
import { db } from "@/db/database";
import type { Document, Category, SourceType } from "@/types/document";
import { uuid } from "@/utils/uuid";

export async function createDocument(input: {
  title: string;
  content: string;
  sourceType?: SourceType;
  sourceName?: string;
  categoryId?: string;
  tags?: string[];
}): Promise<Document> {
  const now = new Date().toISOString();
  const doc: Document = {
    id: uuid(),
    title: input.title.trim(),
    content: input.content,
    sourceType: input.sourceType ?? "note",
    sourceName: input.sourceName,
    categoryId: input.categoryId,
    tags: input.tags ?? [],
    keywords: [],
    favorite: false,
    createdAt: now,
    updatedAt: now
  };
  await db.documents.add(doc);
  return doc;
}

export async function updateDocument(id: string, patch: Partial<Document>): Promise<void> {
  await db.documents.update(id, { ...patch, updatedAt: new Date().toISOString() });
}

export async function deleteDocument(id: string): Promise<void> {
  await db.documents.delete(id);
}

export async function listDocuments(): Promise<Document[]> {
  return db.documents.orderBy("updatedAt").reverse().toArray();
}

export async function getDocument(id: string): Promise<Document | undefined> {
  return db.documents.get(id);
}

export async function toggleFavorite(id: string): Promise<void> {
  const doc = await db.documents.get(id);
  if (doc) await db.documents.update(id, { favorite: !doc.favorite });
}
```

### 7.2 categoryService.ts（同层）

```ts
import { db } from "@/db/database";
import type { Category } from "@/types/document";
import { uuid } from "@/utils/uuid";

export async function createCategory(name: string, color?: string): Promise<Category> {
  const count = await db.categories.count();
  const cat: Category = {
    id: uuid(),
    name: name.trim(),
    color,
    sortOrder: count,
    createdAt: new Date().toISOString()
  };
  await db.categories.add(cat);
  return cat;
}

export async function listCategories(): Promise<Category[]> {
  return db.categories.orderBy("sortOrder").toArray();
}

export async function renameCategory(id: string, name: string): Promise<void> {
  await db.categories.update(id, { name: name.trim() });
}

export async function deleteCategory(id: string): Promise<void> {
  await db.categories.delete(id);
  // 将属于该分类的资料 categoryId 置空
  const docs = await db.documents.where("categoryId").equals(id).toArray();
  await Promise.all(docs.map((d) => db.documents.update(d.id, { categoryId: undefined })));
}
```

## 8. 状态管理（src/stores/）

### 8.1 documentStore.ts

```ts
import { defineStore } from "pinia";
import type { Document, Category } from "@/types/document";
import * as docService from "@/services/documentService";
import * as catService from "@/services/categoryService";

export const useDocumentStore = defineStore("document", {
  state: () => ({
    documents: [] as Document[],
    categories: [] as Category[],
    loading: false,
    searchKeyword: ""
  }),
  getters: {
    favoriteCount: (s) => s.documents.filter((d) => d.favorite).length,
    recentDocuments: (s) => s.documents.slice(0, 5)
  },
  actions: {
    async loadAll() {
      this.loading = true;
      try {
        this.documents = await docService.listDocuments();
        this.categories = await catService.listCategories();
      } finally {
        this.loading = false;
      }
    },
    async create(input: Parameters<typeof docService.createDocument>[0]) {
      const doc = await docService.createDocument(input);
      await this.loadAll();
      return doc;
    },
    async update(id: string, patch: Partial<Document>) {
      await docService.updateDocument(id, patch);
      await this.loadAll();
    },
    async remove(id: string) {
      await docService.deleteDocument(id);
      await this.loadAll();
    },
    async toggleFavorite(id: string) {
      await docService.toggleFavorite(id);
      await this.loadAll();
    }
  }
});
```

通用确认弹窗，接收 `visible`、`title`、`message`、`confirmText`，emit `confirm` / `cancel`。删除资料、批量清理前必须使用。
