## 2. 全文搜索（src/services/searchService.ts）

### 2.1 分词策略

中文采用字符 bigram + 单字混合索引，英文保留单词边界：

```ts
export function tokenize(text: string): string[] {
  const normalized = text.toLowerCase();
  const tokens: string[] = [];

  // 英文单词
  const words = normalized.match(/[a-z0-9]+/g) ?? [];
  tokens.push(...words);

  // 中文字符 bigram
  const cjk = normalized.replace(/[^\u4e00-\u9fa5]/g, "");
  for (let i = 0; i < cjk.length; i++) {
    tokens.push(cjk[i]);
    if (i + 1 < cjk.length) tokens.push(cjk.slice(i, i + 2));
  }
  return [...new Set(tokens)];
}
```

### 2.2 文档匹配与打分

```ts
import type { Document } from "@/types/document";

export interface SearchResult {
  document: Document;
  score: number;
  matchedFields: ("title" | "content" | "tags")[];
  excerpt?: string;
}

export function searchDocuments(docs: Document[], keyword: string): SearchResult[] {
  const kw = keyword.trim().toLowerCase();
  if (!kw) return [];

  const kwTokens = tokenize(kw);
  const results: SearchResult[] = [];

  for (const doc of docs) {
    let score = 0;
    const matchedFields: SearchResult["matchedFields"] = [];

    if (doc.title.toLowerCase().includes(kw)) {
      score += 100;
      matchedFields.push("title");
    }
    if (doc.tags.some((t) => t.toLowerCase().includes(kw))) {
      score += 50;
      matchedFields.push("tags");
    }
    const idx = doc.content.toLowerCase().indexOf(kw);
    if (idx >= 0) {
      score += 30;
      matchedFields.push("content");
      results.push({ document: doc, score, matchedFields, excerpt: makeExcerpt(doc.content, idx, kw.length) });
      continue;
    }
    // 无精确命中时按 token 匹配打分
    const hit = kwTokens.some((t) => doc.content.toLowerCase().includes(t));
    if (hit) {
      score += 10;
      matchedFields.push("content");
    }
    if (score > 0) {
      results.push({ document: doc, score, matchedFields });
    }
  }
  return results.sort((a, b) => b.score - a.score);
}

function makeExcerpt(content: string, idx: number, len: number): string {
  const start = Math.max(0, idx - 30);
  const end = Math.min(content.length, idx + len + 40);
  return (start > 0 ? "…" : "") + content.slice(start, end) + (end < content.length ? "…" : "");
}
```

### 2.3 高亮组件（HighlightText.vue）

```vue
<script setup lang="ts">
import { computed } from "vue";
const props = defineProps<{ text: string; keyword: string }>();
const parts = computed(() => {
  if (!props.keyword) return [{ text: props.text, hit: false }];
  const lower = props.text.toLowerCase();
  const kw = props.keyword.toLowerCase();
  const res: { text: string; hit: boolean }[] = [];
  let i = 0;
  while (i < lower.length) {
    const idx = lower.indexOf(kw, i);
    if (idx < 0) { res.push({ text: props.text.slice(i), hit: false }); break; }
    if (idx > i) res.push({ text: props.text.slice(i, idx), hit: false });
    res.push({ text: props.text.slice(idx, idx + kw.length), hit: true });
    i = idx + kw.length;
  }
  return res;
});
</script>

<template>
  <span>
    <mark v-for="(p, i) in parts" :key="i" :class="p.hit ? 'bg-yellow-200 rounded px-0.5' : ''">{{ p.text }}</mark>
  </span>
</template>
```

> 说明：MVP 使用本地内存检索即可（文档量级小），无需倒排索引。若后续数据量大，可在此 Service 内替换为倒排索引或后端检索，页面无感知。

## 3. 搜索与筛选接入（LibraryPage）

在 `documentStore` 增加 computed `filteredDocuments`：

```ts
getters: {
  filteredDocuments(state): Document[] {
    let list = searchDocuments(state.documents, state.searchKeyword)
      .map((r) => r.document);
    // 分类筛选
    if (state.activeCategoryId) {
      list = list.filter((d) => d.categoryId === state.activeCategoryId);
    }
    // 收藏筛选
    if (state.onlyFavorite) {
      list = list.filter((d) => d.favorite);
    }
    return list;
  }
}
```

搜索框实时筛选：`SearchBar` 的 `@input` 更新 `searchKeyword`，列表自动重算。清空按钮清空关键词与筛选条件。

## 6. 任务管理（src/services/taskService.ts + src/stores/taskStore.ts）

### 6.1 taskService.ts

```ts
import { db } from "@/db/database";
import type { Task, TaskStatus, TaskPriority } from "@/types/task";
import { uuid } from "@/utils/uuid";

export async function createTask(input: {
  title: string;
  description?: string;
  sourceDocumentId?: string;
  priority?: TaskPriority;
  dueDate?: string;
}): Promise<Task> {
  const now = new Date().toISOString();
  const task: Task = {
    id: uuid(),
    title: input.title.trim(),
    description: input.description,
    sourceDocumentId: input.sourceDocumentId,
    status: "todo",
    priority: input.priority ?? "medium",
    dueDate: input.dueDate,
    createdAt: now,
    updatedAt: now
  };
  await db.tasks.add(task);
  return task;
}

export async function listTasks(filter?: { status?: TaskStatus }): Promise<Task[]> {
  if (filter?.status) return db.tasks.where("status").equals(filter.status).toArray();
  return db.tasks.toArray();
}

export async function updateTaskStatus(id: string, status: TaskStatus): Promise<void> {
  await db.tasks.update(id, { status, updatedAt: new Date().toISOString() });
}

export async function updateTask(id: string, patch: Partial<Task>): Promise<void> {
  await db.tasks.update(id, { ...patch, updatedAt: new Date().toISOString() });
}

export async function deleteTask(id: string): Promise<void> {
  await db.tasks.delete(id);
}
```

### 6.2 taskStore.ts

```ts
export const useTaskStore = defineStore("task", {
  state: () => ({ tasks: [] as Task[], filterStatus: "all" as "all" | TaskStatus }),
  getters: {
    filteredTasks(state): Task[] {
      if (state.filterStatus === "all") return state.tasks;
      return state.tasks.filter((t) => t.status === state.filterStatus);
    }
  },
  actions: {
    async load() { this.tasks = await listTasks(); },
    async create(input) { const t = await createTask(input); await this.load(); return t; },
    async setStatus(id, status) { await updateTaskStatus(id, status); await this.load(); },
    async update(id, patch) { await updateTask(id, patch); await this.load(); },
    async remove(id) { await deleteTask(id); await this.load(); }
  }
});
```

## 7. 学习卡片（src/services/flashcardService.ts + StudyPage.vue）

### 7.1 flashcardService.ts

```ts
import { db } from "@/db/database";
import { uuid } from "@/utils/uuid";

export type Mastery = 0 | 1 | 2;

export interface Flashcard {
  id: string;
  documentId: string;
  question: string;
  answer: string;
  mastery: Mastery;
  reviewCount: number;
  nextReviewAt?: string;
  createdAt: string;
}

export async function listFlashcards(): Promise<Flashcard[]> {
  return db.flashcards.toArray();
}

export async function createFlashcard(input: {
  documentId: string;
  question: string;
  answer: string;
}): Promise<Flashcard> {
  const card: Flashcard = {
    id: uuid(),
    documentId: input.documentId,
    question: input.question,
    answer: input.answer,
    mastery: 0,
    reviewCount: 0,
    createdAt: new Date().toISOString()
  };
  await db.flashcards.add(card);
  return card;
}

export async function markMastery(id: string, mastery: Mastery): Promise<void> {
  const card = await db.flashcards.get(id);
  if (!card) return;
  const intervals = [1, 3, 7]; // 天数（简单复习间隔）
  const nextReviewAt = new Date(Date.now() + intervals[mastery] * 24 * 3600 * 1000).toISOString();
  await db.flashcards.update(id, {
    mastery,
    reviewCount: card.reviewCount + 1,
    nextReviewAt
  });
}
```
